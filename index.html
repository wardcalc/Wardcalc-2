'use strict'; // --- 1. MEMORY & AUTO-TRANSLATE ON LOAD --- document.addEventListener('DOMContentLoaded', () => { let savedLang = localStorage.getItem('wardcalc_lang') || 'en'; // Set language globally if variables exist if (typeof LANG !== 'undefined') { LANG = savedLang; } // Highlight the correct button instantly document.querySelectorAll('.lang-btn').forEach(btn => { if (btn.innerText.toLowerCase() === savedLang.toLowerCase()) { btn.classList.add('active'); btn.style.background = 'var(--gold)'; btn.style.color = '#000'; btn.style.border = 'none'; } else { btn.classList.remove('active'); btn.style.background = 'transparent'; btn.style.color = 'var(--w60)'; btn.style.border = 'none'; } }); // Run translation if (typeof updateUI === 'function') { updateUI(); } }); // --- 2. SWITCH LANGUAGE --- function setLang(lang, btn) { localStorage.setItem('wardcalc_lang', lang); if (typeof LANG !== 'undefined') LANG = lang; document.querySelectorAll('.lang-btn').forEach(b => { b.classList.remove('active'); b.style.background = 'transparent'; b.style.color = 'var(--w60)'; }); if (btn) { btn.classList.add('active'); btn.style.background = 'var(--gold)'; btn.style.color = '#000'; } if (typeof updateUI === 'function') updateUI(); } // --- 3. THE TRANSLATOR --- function updateUI() { if (typeof t !== 'function') return; document.querySelectorAll('[data-k]').forEach(el => { const key = el.getAttribute('data-k'); const translatedText = t(key); if (translatedText !== key) { if (el.tagName === 'INPUT' && el.hasAttribute('placeholder')) { el.placeholder = translatedText; } else { el.innerHTML = translatedText; } } }); } // --- 4. BUTTON CLICK FIXES --- // For Yes/No rows (Wells, Child-Pugh) function pickChip(el) { const group = el.getAttribute('data-g'); if (!group) return; document.querySelectorAll(`[data-g="${group}"]`).forEach(b => b.classList.remove('on')); el.classList.add('on'); } // For multi-select / PSI PORT score toggles function toggleChip(el) { el.classList.toggle('on'); } // --- 5. SEARCH BAR --- function filterTools() { let input = document.getElementById('search'); if (!input) return; let filter = input.value.toLowerCase(); let tools = document.querySelectorAll('.tool-card'); tools.forEach(tool => { if (tool.innerText.toLowerCase().includes(filter)) { tool.style.display = "block"; } else { tool.style.display = "none"; } }); }
